number of parameters: 0
fields:

constructor:

    Blocked1DLaunch
def VeriTile.Triton.Blocked1DLaunch.gridX : Blocked1DLaunch → Nat :=
fun c => c.grid.headD 0
def VeriTile.Triton.Blocked1DLaunch.bufs : Blocked1DLaunch → List BufMeta :=
fun c => c.output :: c.inputs
def VeriTile.Triton.Blocked1DLaunch.i32Limit : Nat :=
2 ^ 31
def VeriTile.Triton.Blocked1DLaunch.RangesDisjoint : Nat → BufMeta → BufMeta → Prop :=
fun n a b =>
  ∀ (addr : Nat), ¬(a.base ≤ addr ∧ addr < a.base + n * a.elemBytes ∧ b.base ≤ addr ∧ addr < b.base + n * b.elemBytes)
def VeriTile.Triton.Blocked1DLaunch.GridRank : Blocked1DLaunch → Prop :=
fun c => c.grid.length = 1
def VeriTile.Triton.Blocked1DLaunch.BlockOk : Blocked1DLaunch → Prop :=
fun c => ∃ k, k ≤ 20 ∧ c.block = 2 ^ k
def VeriTile.Triton.Blocked1DLaunch.Covers : Blocked1DLaunch → Prop :=
fun c => ∀ (i : Nat), i < c.n → i / c.block < c.gridX
def VeriTile.Triton.Blocked1DLaunch.LanesInBounds : Blocked1DLaunch → Prop :=
fun c =>
  ∀ (b : BufMeta),
    b ∈ c.bufs →
      ∀ (pid : Nat), pid < c.gridX → ∀ (j : Nat), j < c.block → pid * c.block + j < c.n → pid * c.block + j < b.capacity
def VeriTile.Triton.Blocked1DLaunch.OffsetsFit : Blocked1DLaunch → Prop :=
fun c => ∀ (pid : Nat), pid < c.gridX → ∀ (j : Nat), j < c.block → pid * c.block + j < Blocked1DLaunch.i32Limit
def VeriTile.Triton.Blocked1DLaunch.NFits : Blocked1DLaunch → Prop :=
fun c => c.n < Blocked1DLaunch.i32Limit
def VeriTile.Triton.Blocked1DLaunch.GridFits : Blocked1DLaunch → Prop :=
fun c => c.gridX < Blocked1DLaunch.i32Limit
def VeriTile.Triton.Blocked1DLaunch.UnitStride : Blocked1DLaunch → Prop :=
fun c => ∀ (b : BufMeta), b ∈ c.bufs → c.n ≤ 1 ∨ b.stride = 1
def VeriTile.Triton.Blocked1DLaunch.DTypeOk : Blocked1DLaunch → Prop :=
fun c => ∀ (b : BufMeta), b ∈ c.bufs → b.dtype = ElemDType.f32 ∧ b.elemBytes = 4
def VeriTile.Triton.Blocked1DLaunch.OutDisjoint : Blocked1DLaunch → Prop :=
fun c => ∀ (b : BufMeta), b ∈ c.inputs → Blocked1DLaunch.RangesDisjoint c.n c.output b

  c.Pre → ∀ {p₁ p₂ j₁ j₂ : Nat}, j₁ < c.block → j₂ < c.block → p₁ * c.block + j₁ = p₂ * c.block + j₂ → p₁ = p₂ ∧ j₁ = j₂
@Blocked1DLaunch.Pre.i32_offset_toInt : ∀ {c : Blocked1DLaunch},
  c.Pre →
    ∀ {pid j : Nat},
      pid < c.gridX →
        j < c.block → (BitVec.ofNat 32 pid * BitVec.ofNat 32 c.block + BitVec.ofNat 32 j).toInt = ↑(pid * c.block + j)
@Blocked1DLaunch.Pre.i32_mask_eq : ∀ {c : Blocked1DLaunch},
  c.Pre →
    ∀ {pid j : Nat},
      pid < c.gridX →
        j < c.block →
          (BitVec.ofNat 32 pid * BitVec.ofNat 32 c.block + BitVec.ofNat 32 j).slt (BitVec.ofNat 32 c.n) =
            decide (pid * c.block + j < c.n)
