def VeriTile.Triton.FastDiv.shiftFor : ℕ → ℕ :=
fun d => VeriTile.Triton.FastDiv.shiftFor.go d 0 33
def VeriTile.Triton.FastDiv.magic : ℕ → ℕ :=
fun d => 2 ^ 32 * (2 ^ VeriTile.Triton.FastDiv.shiftFor d - d) / d + 1
def VeriTile.Triton.FastDiv.quot : ℕ → ℕ → ℕ :=
fun d x => (x * VeriTile.Triton.FastDiv.magic d / 2 ^ 32 + x) / 2 ^ VeriTile.Triton.FastDiv.shiftFor d
def VeriTile.Triton.FastDiv.umulhi32 : BitVec 32 → BitVec 32 → BitVec 32 :=
fun a b => BitVec.ofNat 32 (a.toNat * b.toNat / 2 ^ 32)
@VeriTile.Triton.FastDiv.shiftFor_spec : ∀ {d : ℕ},
  0 < d →
    d ≤ 2 ^ 32 →
      d ≤ 2 ^ VeriTile.Triton.FastDiv.shiftFor d ∧
        (VeriTile.Triton.FastDiv.shiftFor d = 0 ∨ 2 ^ (VeriTile.Triton.FastDiv.shiftFor d - 1) < d)
@VeriTile.Triton.FastDiv.magic_quotient : ∀ {d x : ℕ},
  0 < d → d < 2 ^ 31 → x < 2 ^ 31 → VeriTile.Triton.FastDiv.quot d x = x / d

def VeriTile.Bench.Optimizations.CatRepack.segCat : List (ℕ × (ℕ → ℝ)) → ℕ → ℕ → ℝ :=
fun x x_1 x_2 =>
  List.brecOn (motive := fun x => ℕ → ℕ → ℝ) x
    (fun x f x_3 x_4 =>
      (match (motive := (x : List (ℕ × (ℕ → ℝ))) → ℕ → ℕ → List.below (motive := fun x => ℕ → ℕ → ℝ) x → ℝ) x, x_3,
          x_4 with
        | [], x, x_5 => fun x => 0
        | (w, f) :: rest, r, c => fun x => if c < w then f (r * w + c) else x.1 r (c - w))
        f)
    x_1 x_2

  (q1 k1 va1 vb1 q2 k2 va2 vb2 out : VeriTile.Triton.RegionName) (n wq wk wv BLOCK : ℕ),

        have W := 2 * (wq + wk + wv);

            out BLOCK s.pid i =

                (s.readMem vb1) (s.readMem q2) (s.readMem k2) (s.readMem va2) (s.readMem vb2) (x / W) (x % W)
