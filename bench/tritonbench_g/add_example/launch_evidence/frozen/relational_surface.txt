def VeriTile.Triton.Blocked1D.ProgramRuns : VeriTile.Triton.Kernel →
  VeriTile.Triton.RegionName → ℕ → ℕ → (VeriTile.Triton.BlockState → Prop) → (ℕ → ℝ) → Prop :=
fun k R n B pre F =>
  ∀ (t : VeriTile.Triton.BlockState),
    pre t →
      ∃ f,
        VeriTile.Triton.exec k t = some f ∧
          (∀ o < n, o / B = t.pid → f.readMem R o = F o) ∧
            ∀ (r : VeriTile.Triton.RegionName) (o : ℕ),
              ¬VeriTile.Triton.Blocked1D.blockWrites R n B t.pid (r, o) → f.mem r o = t.mem r o
@VeriTile.Triton.Blocked1D.launch_of_programRuns : ∀ {k : VeriTile.Triton.Kernel} {R : VeriTile.Triton.RegionName}
  {n B g : ℕ} {pre : VeriTile.Triton.BlockState → Prop} {F : ℕ → ℝ},
  0 < B →
    n ≤ g * B →
      ∀ (s : VeriTile.Triton.BlockState),
        VeriTile.Triton.Blocked1D.ProgramRuns k R n B pre F →
          (∀ (idx : VeriTile.Triton.GridIndex (VeriTile.Triton.Blocked1D.line g)), pre (s.withGridIndex idx)) →
            ∃ sF,
              Nonempty (k.GridLaunchedOrdinary (VeriTile.Triton.Blocked1D.line g) s sF) ∧
                (∀ o < n, sF.readMem R o = F o) ∧
                  ∀ (r : VeriTile.Triton.RegionName) (o : ℕ), ¬(r = R ∧ o < n) → sF.mem r o = s.mem r o
@VeriTile.Triton.Blocked1D.fused_agrees_two_launch : ∀ {kA kB kF : VeriTile.Triton.Kernel}
  {T O : VeriTile.Triton.RegionName} {n B g : ℕ},
  0 < B →
    n ≤ g * B →
      ∀ {preA preF : VeriTile.Triton.BlockState → Prop} {F : ℕ → ℝ} {G : ℝ → ℝ} (s : VeriTile.Triton.BlockState),
        VeriTile.Triton.Blocked1D.ProgramRuns kA T n B preA F →
          (∀ (idx : VeriTile.Triton.GridIndex (VeriTile.Triton.Blocked1D.line g)), preA (s.withGridIndex idx)) →
            (VeriTile.Triton.Blocked1D.ProgramRuns kB O n B (fun t => ∀ o < n, t.readMem T o = F o) fun o => G (F o)) →
              (VeriTile.Triton.Blocked1D.ProgramRuns kF O n B preF fun o => G (F o)) →
                (∀ (idx : VeriTile.Triton.GridIndex (VeriTile.Triton.Blocked1D.line g)), preF (s.withGridIndex idx)) →
                  ∃ sT sU sF,
                    Nonempty (kA.GridLaunchedOrdinary (VeriTile.Triton.Blocked1D.line g) s sT) ∧
                      Nonempty (kB.GridLaunchedOrdinary (VeriTile.Triton.Blocked1D.line g) sT sU) ∧
                        Nonempty (kF.GridLaunchedOrdinary (VeriTile.Triton.Blocked1D.line g) s sF) ∧
                          (∀ o < n, sF.readMem O o = sU.readMem O o) ∧
                            (∀ o < n, sF.readMem O o = G (F o)) ∧
                              ∀ (r : VeriTile.Triton.RegionName) (o : ℕ),
                                ¬(r = T ∧ o < n) → ¬(r = O ∧ o < n) → sF.mem r o = sU.mem r o

VeriTile.Bench.TritonBenchG.AddReluRelational.add_relu_fusion_relational : ∀ (c : VeriTile.Triton.Blocked1DLaunch),
  c.check = «true» →
    ∀ (x y T O : VeriTile.Triton.RegionName) (s : VeriTile.Triton.BlockState) (xs ys : ℕ → ℝ),
      (∀ o < c.n, s.readMem x o = xs o) →
        (∀ o < c.n, s.readMem y o = ys o) →
          ∃ sT sU sF,
            Nonempty
                (VeriTile.Triton.Kernel.GridLaunchedOrdinary
                  (VeriTile.Bench.TritonBenchG.AddReluRelational.add_kernel x y T c.n c.block).toAlgKernel
                  { dims := c.grid } s sT) ∧
              Nonempty
                  (VeriTile.Triton.Kernel.GridLaunchedOrdinary
                    (VeriTile.Bench.TritonBenchG.AddReluRelational.relu_kernel T O c.n c.block).toAlgKernel
                    { dims := c.grid } sT sU) ∧
                Nonempty
                    (VeriTile.Triton.Kernel.GridLaunchedOrdinary
                      (VeriTile.Bench.TritonBenchG.AddReluRelational.add_relu_kernel x y O c.n c.block).toAlgKernel
                      { dims := c.grid } s sF) ∧
                  (∀ o < c.n, sF.readMem O o = sU.readMem O o) ∧
                    (∀ o < c.n, sF.readMem O o = VeriTile.Triton.TiledActivation.relu (xs o + ys o)) ∧
                      ∀ (r : VeriTile.Triton.RegionName) (o : ℕ),
                        ¬(r = T ∧ o < c.n) → ¬(r = O ∧ o < c.n) → sF.mem r o = sU.mem r o
