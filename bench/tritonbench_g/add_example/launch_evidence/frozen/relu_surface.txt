def VeriTile.Triton.StridedUnary.nextPow2Aux : ℕ → ℕ → ℕ → ℕ :=
fun x x_1 x_2 =>
  Nat.brecOn (motive := fun x => ℕ → ℕ → ℕ) x
    (fun x f x_3 x_4 =>
      (match (motive := (x : ℕ) → ℕ → ℕ → Nat.below (motive := fun x => ℕ → ℕ → ℕ) x → ℕ) x, x_3, x_4 with
        | 0, p, x => fun x => p
        | fuel.succ, p, n => fun x => if n ≤ p then p else x.1 (2 * p) n)
        f)
    x_1 x_2
