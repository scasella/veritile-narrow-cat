def VeriTile.Bench.TritonBenchG.AddReluFused.addReluIO : VeriTile.Triton.RegionName →
  VeriTile.Triton.RegionName → VeriTile.Triton.RegionName → ℕ → ℕ → VeriTile.Triton.MaskedKernelIO₂ :=
fun in_ptr0 in_ptr1 out_ptr n_elements BLOCK_SIZE =>
  { kernel := VeriTile.Bench.TritonBenchG.AddReluFused.add_relu_kernel in_ptr0 in_ptr1 out_ptr n_elements BLOCK_SIZE,
    projection := ⋯, in1 := in_ptr0, in2 := in_ptr1, out := out_ptr, B := BLOCK_SIZE,
    read1 := fun pid => pid * BLOCK_SIZE, read2 := fun pid => pid * BLOCK_SIZE, write := fun pid => pid * BLOCK_SIZE,
    mask := fun pid j => pid * BLOCK_SIZE + ↑j < n_elements }
def VeriTile.Triton.MaskedKernelIO₂.Implements : (io : VeriTile.Triton.MaskedKernelIO₂) →
  ((Fin io.B → ℝ) → (Fin io.B → ℝ) → Fin io.B → ℝ) → Prop :=
fun io f =>
  ∀ (A : VeriTile.Triton.FlatAlloc),
    A.Disjoint →
      A.regions = [io.in1, io.in2, io.out] ++ List.map Prod.fst io.scratch →
        (∀ r ∉ A.regions, A.extent r = 0) →
          ∀ (pid : ℕ),
            (∀ (j : Fin io.B), io.mask pid j → io.read1 pid + ↑j < A.extent io.in1) →
              (∀ (j : Fin io.B), io.mask pid j → io.read2 pid + ↑j < A.extent io.in2) →
                (∀ (j : Fin io.B), io.mask pid j → io.write pid + ↑j < A.extent io.out) →
                  (∀ p ∈ io.scratch, ∀ (j : Fin io.B), io.mask pid j → p.2 pid + ↑j < A.extent p.1) →
                    ∀ (xs ys : Fin io.B → ℝ) (s₀ : VeriTile.Triton.BlockState),
                      s₀.pid = pid →
                        (s₀.undef = fun x x_1 => 0) →
                          (∀ (j : Fin io.B), io.mask pid j → s₀.readMem io.in1 (io.read1 pid + ↑j) = xs j) →
                            (∀ (j : Fin io.B), io.mask pid j → s₀.readMem io.in2 (io.read2 pid + ↑j) = ys j) →
                              ∃ s',
                                VeriTile.Triton.exec (A.flattenKernel io.kernel.toAlgKernel) (A.flattenState s₀) =
                                    some s' ∧
                                  (∀ (j : Fin io.B),
                                      io.mask pid j →
                                        s'.readMem A.flat (A.addr io.out (io.write pid + ↑j)) = f xs ys j) ∧
                                    ∀ (r' : VeriTile.Triton.RegionName) (o' : ℕ),
                                      (r' ≠ A.flat ∨
                                          (∀ (j : Fin io.B), io.mask pid j → o' ≠ A.addr io.out (io.write pid + ↑j)) ∧
                                            ∀ p ∈ io.scratch,
                                              ∀ (j : Fin io.B), io.mask pid j → o' ≠ A.addr p.1 (p.2 pid + ↑j)) →
                                        s'.mem r' o' = (A.flattenState s₀).mem r' o'

def VeriTile.Triton.Elementwise2.flatAlloc : VeriTile.Triton.RegionName →
  VeriTile.Triton.RegionName →
    VeriTile.Triton.RegionName →
      VeriTile.Triton.RegionName →
        VeriTile.Triton.TensorMeta →
          VeriTile.Triton.TensorMeta → VeriTile.Triton.TensorMeta → VeriTile.Triton.FlatAlloc :=
fun flat in0 in1 outR x y out =>
  { flat := flat, regions := [in0, in1, outR],
    base := fun r =>
      if r = outR then out.base / out.elemBytes
      else if r = in0 then x.base / x.elemBytes else if r = in1 then y.base / y.elemBytes else 0,
    extent := fun r => if r = outR ∨ r = in0 ∨ r = in1 then x.numel else 0 }
