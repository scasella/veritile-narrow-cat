def VeriTile.Triton.Kernel.LaunchCorrectFramed : {ι : Type} →
  VeriTile.Triton.Kernel →
    VeriTile.Triton.Grid → VeriTile.Triton.BlockState → (ι → Option VeriTile.Triton.MemCellAddr) → (ι → ℝ) → Prop :=
fun {ι} k g s write expected =>
  ∃ sFinal x,
    (∀ (i : ι) (addr : VeriTile.Triton.MemCellAddr), write i = some addr → sFinal.readMem addr.1 addr.2 = expected i) ∧
      ∀ (addr : VeriTile.Triton.MemCellAddr),
        (∀ (i : ι), write i ≠ some addr) → sFinal.mem addr.1 addr.2 = s.mem addr.1 addr.2

def VeriTile.Triton.Kernel.mergeFrames : {k : VeriTile.Triton.Kernel} →
  (g : VeriTile.Triton.Grid) → (s : VeriTile.Triton.BlockState) → k.GridFrames g s → VeriTile.Triton.BlockState :=
fun {k} g s frames =>
  {
    mem := fun region offset =>
      if h : ∃ idx, (frames idx).writes (region, offset) then (frames (Classical.choose h)).final.mem region offset
      else s.mem region offset,
    regs := s.regs, pids := s.pids, undef := s.undef, numPids := s.numPids }

def VeriTile.Triton.MaskedKernelIO₂.Implements : (io : VeriTile.Triton.MaskedKernelIO₂) →
  ((Fin io.B → ℝ) → (Fin io.B → ℝ) → Fin io.B → ℝ) → Prop :=
fun io f =>
  ∀ (A : VeriTile.Triton.FlatAlloc),
    A.Disjoint →
      A.regions = [io.in1, io.in2, io.out] ++ List.map Prod.fst io.scratch →
        (∀ r ∉ A.regions, A.extent r = 0) →
          ∀ (pid : ℕ),
            (∀ (j : Fin io.B), io.mask pid j → io.read1 pid + ↑j < A.extent io.in1) →
              (∀ (j : Fin io.B), io.mask pid j → io.read2 pid + ↑j < A.extent io.in2) →
                (∀ (j : Fin io.B), io.mask pid j → io.write pid + ↑j < A.extent io.out) →
                  (∀ p ∈ io.scratch, ∀ (j : Fin io.B), io.mask pid j → p.2 pid + ↑j < A.extent p.1) →
                    ∀ (xs ys : Fin io.B → ℝ) (s₀ : VeriTile.Triton.BlockState),
                      s₀.pid = pid →
                        (s₀.undef = fun x x_1 => 0) →
                          (∀ (j : Fin io.B), io.mask pid j → s₀.readMem io.in1 (io.read1 pid + ↑j) = xs j) →
                            (∀ (j : Fin io.B), io.mask pid j → s₀.readMem io.in2 (io.read2 pid + ↑j) = ys j) →
                              ∃ s',
                                VeriTile.Triton.exec (A.flattenKernel io.kernel.toAlgKernel) (A.flattenState s₀) =
                                    some s' ∧
                                  (∀ (j : Fin io.B),
                                      io.mask pid j →
                                        s'.readMem A.flat (A.addr io.out (io.write pid + ↑j)) = f xs ys j) ∧
                                    ∀ (r' : VeriTile.Triton.RegionName) (o' : ℕ),
                                      (r' ≠ A.flat ∨
                                          (∀ (j : Fin io.B), io.mask pid j → o' ≠ A.addr io.out (io.write pid + ↑j)) ∧
                                            ∀ p ∈ io.scratch,
                                              ∀ (j : Fin io.B), io.mask pid j → o' ≠ A.addr p.1 (p.2 pid + ↑j)) →
                                        s'.mem r' o' = (A.flattenState s₀).mem r' o'

      ∀ {s : VeriTile.Triton.BlockState} {out : VeriTile.Triton.RegionName}
        (frames : k.GridFrames (VeriTile.Triton.Blocked1D.line g) s),
        (∀ (idx : VeriTile.Triton.GridIndex (VeriTile.Triton.Blocked1D.line g)),
            (frames idx).writes = VeriTile.Triton.Blocked1D.blockWrites out n B (VeriTile.Triton.Blocked1D.pidOf idx)) →
          ∀ (expected : ℕ → ℝ),
            (∀ (idx : VeriTile.Triton.GridIndex (VeriTile.Triton.Blocked1D.line g)),
                ∀ j < B,
                  VeriTile.Triton.Blocked1D.pidOf idx * B + j < n →
                    (frames idx).final.readMem out (VeriTile.Triton.Blocked1D.pidOf idx * B + j) =
                      expected (VeriTile.Triton.Blocked1D.pidOf idx * B + j)) →
              ∃ L,
                L.frames = frames ∧
                  (∀ i < n,
                      (VeriTile.Triton.Kernel.mergeFrames (VeriTile.Triton.Blocked1D.line g) s frames).readMem out i =
                        expected i) ∧
                    ∀ (r : VeriTile.Triton.RegionName) (o : ℕ),
                      ¬(r = out ∧ o < n) →
                        (VeriTile.Triton.Kernel.mergeFrames (VeriTile.Triton.Blocked1D.line g) s frames).mem r o =
                          s.mem r o
