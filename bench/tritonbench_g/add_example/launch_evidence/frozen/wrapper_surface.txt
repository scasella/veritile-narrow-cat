def VeriTile.Triton.TensorMeta.numel : VeriTile.Triton.TensorMeta → ℕ :=
fun t => List.foldr (fun x1 x2 => x1 * x2) 1 t.shape
def VeriTile.Triton.TensorMeta.rowMajor : List ℕ → List ℕ :=
fun x =>
  List.brecOn x fun x f =>
    (match (motive := (x : List ℕ) → List.below x → List ℕ) x with
      | [] => fun x => []
      | head :: rest => fun x => List.foldr (fun x1 x2 => x1 * x2) 1 rest :: x.1)
      f
def VeriTile.Triton.TensorMeta.Contiguous : VeriTile.Triton.TensorMeta → Prop :=
fun t =>
  t.strides.length = t.shape.length ∧
    ∀ i < t.shape.length,
      1 < t.shape.getD i 0 → t.strides.getD i 0 = (VeriTile.Triton.TensorMeta.rowMajor t.shape).getD i 0
def VeriTile.Triton.TensorMeta.offsetOf : VeriTile.Triton.TensorMeta → List ℕ → ℕ :=
fun t idx => List.foldr (fun p acc => p.1 * p.2 + acc) 0 (idx.zip t.strides)
def VeriTile.Triton.TensorMeta.InShape : List ℕ → List ℕ → Prop :=
fun shape idx => idx.length = shape.length ∧ ∀ k < shape.length, idx.getD k 0 < shape.getD k 0
def VeriTile.Triton.TensorMeta.Aligned : VeriTile.Triton.TensorMeta → Prop :=
fun t => t.base % t.elemBytes = 0
def VeriTile.Triton.TensorMeta.toBuf : VeriTile.Triton.TensorMeta → VeriTile.Triton.BufMeta :=
fun t =>
  { base := t.base, elemBytes := t.elemBytes, stride := if t.contiguousB = «true» then 1 else 0, capacity := t.capacity,
    dtype := t.dtype }
def VeriTile.Triton.TensorMeta.linear : List ℕ → List ℕ → ℕ :=
fun x x_1 =>
  List.brecOn (motive := fun x => List ℕ → ℕ) x
    (fun x f x_2 =>
      (match (motive := (x : List ℕ) → List ℕ → List.below (motive := fun x => List ℕ → ℕ) x → ℕ) x, x_2 with
        | head :: rest, i :: is => fun x => i * List.foldr (fun x1 x2 => x1 * x2) 1 rest + x.1 is
        | x, x_3 => fun x => 0)
        f)
    x_1

    VeriTile.Triton.TensorMeta → VeriTile.Triton.TensorMeta → VeriTile.Triton.Blocked1DLaunch :=
fun B x y out =>
  { n := x.numel, block := B, grid := [(x.numel + B - 1) / B], inputs := [x.toBuf, y.toBuf], output := out.toBuf }
def VeriTile.Triton.Elementwise2.SameShape : VeriTile.Triton.TensorMeta →
  VeriTile.Triton.TensorMeta → VeriTile.Triton.TensorMeta → Prop :=
fun x y out => y.shape = x.shape ∧ out.shape = x.shape
def VeriTile.Triton.Elementwise2.AllContiguous : VeriTile.Triton.TensorMeta →
  VeriTile.Triton.TensorMeta → VeriTile.Triton.TensorMeta → Prop :=
fun x y out => x.Contiguous ∧ y.Contiguous ∧ out.Contiguous
def VeriTile.Triton.Elementwise2.AllAligned : VeriTile.Triton.TensorMeta →
  VeriTile.Triton.TensorMeta → VeriTile.Triton.TensorMeta → Prop :=
fun x y out => x.Aligned ∧ y.Aligned ∧ out.Aligned
def VeriTile.Triton.Elementwise2.InputsSeparate : VeriTile.Triton.TensorMeta → VeriTile.Triton.TensorMeta → Prop :=
fun x y => x.base = y.base ∨ VeriTile.Triton.Blocked1DLaunch.RangesDisjoint x.numel x.toBuf y.toBuf

def VeriTile.Triton.Elementwise2.flatAlloc : VeriTile.Triton.RegionName →
  VeriTile.Triton.RegionName →
    VeriTile.Triton.RegionName →
      VeriTile.Triton.RegionName →
        VeriTile.Triton.TensorMeta →
          VeriTile.Triton.TensorMeta → VeriTile.Triton.TensorMeta → VeriTile.Triton.FlatAlloc :=
fun flat in0 in1 outR x y out =>
  { flat := flat, regions := [in0, in1, outR],
    base := fun r =>
      if r = outR then out.base / out.elemBytes
      else if r = in0 then x.base / x.elemBytes else if r = in1 then y.base / y.elemBytes else 0,
    extent := fun r => if r = outR ∨ r = in0 ∨ r = in1 then x.numel else 0 }

def VeriTile.Triton.Kernel.runSerial : VeriTile.Triton.Kernel →
  {g : VeriTile.Triton.Grid} →
    VeriTile.Triton.BlockState →
      List (VeriTile.Triton.GridIndex g) →
        (VeriTile.Triton.RegionName → ℕ → VeriTile.Triton.MemCell) →
          Option (VeriTile.Triton.RegionName → ℕ → VeriTile.Triton.MemCell) :=
fun k {g} s x x_1 =>
  List.brecOn (motive := fun x =>
    (VeriTile.Triton.RegionName → ℕ → VeriTile.Triton.MemCell) →
      Option (VeriTile.Triton.RegionName → ℕ → VeriTile.Triton.MemCell))
    x
    (fun x f x_2 =>
      (match (motive :=
          (x : List (VeriTile.Triton.GridIndex g)) →
            (VeriTile.Triton.RegionName → ℕ → VeriTile.Triton.MemCell) →
              List.below (motive := fun x =>
                  (VeriTile.Triton.RegionName → ℕ → VeriTile.Triton.MemCell) →
                    Option (VeriTile.Triton.RegionName → ℕ → VeriTile.Triton.MemCell))
                  x →
                Option (VeriTile.Triton.RegionName → ℕ → VeriTile.Triton.MemCell))
          x, x_2 with
        | [], m => fun x => some m
        | idx :: rest, m => fun x =>
          match
            VeriTile.Triton.exec k
              ({ mem := m, regs := s.regs, pids := s.pids, undef := s.undef, numPids := s.numPids }.withGridIndex
                idx) with
          | none => none
          | some f => x.1 f.mem)
        f)
    x_1
def VeriTile.Triton.Kernel.memReal : (VeriTile.Triton.RegionName → ℕ → VeriTile.Triton.MemCell) →
  VeriTile.Triton.RegionName → ℕ → ℝ :=
fun m r o =>
  match (m r o).readAs VeriTile.Triton.TileDType.real with
  | some (some value) => value
  | x => 0
def VeriTile.Triton.Kernel.FrameRobust : {k : VeriTile.Triton.Kernel} →
  {g : VeriTile.Triton.Grid} → {s : VeriTile.Triton.BlockState} → k.GridFrames g s → Prop :=
fun {k} {g} {s} frames =>
  ∀ (idx : VeriTile.Triton.GridIndex g) (m : VeriTile.Triton.RegionName → ℕ → VeriTile.Triton.MemCell),
    (∀ (r : VeriTile.Triton.RegionName) (o : ℕ),
        (¬∃ idx', idx' ≠ idx ∧ (frames idx').writes (r, o)) → m r o = s.mem r o) →
      ∃ f,
        VeriTile.Triton.exec k
              ({ mem := m, regs := s.regs, pids := s.pids, undef := s.undef, numPids := s.numPids }.withGridIndex idx) =
            some f ∧
          (∀ (r : VeriTile.Triton.RegionName) (o : ℕ),
              (frames idx).writes (r, o) → f.readMem r o = (frames idx).final.readMem r o) ∧
            ∀ (r : VeriTile.Triton.RegionName) (o : ℕ), ¬(frames idx).writes (r, o) → f.mem r o = m r o
